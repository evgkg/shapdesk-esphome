cmake_minimum_required(VERSION 3.20)
project(shapdesk_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_library(desk_core OBJECT components/shapdesk/desk_protocol.cpp components/shapdesk/desk_tracking.cpp components/shapdesk/desk_diagnostics.cpp)
target_include_directories(desk_core PUBLIC components/shapdesk)
foreach(name parser checksum tracking rejections diagnostics control)
  add_executable(test_${name} tests/test_${name}.cpp)
  target_link_libraries(test_${name} PRIVATE desk_core)
  if(MSVC)
    target_compile_options(test_${name} PRIVATE /UNDEBUG)
  else()
    target_compile_options(test_${name} PRIVATE -UNDEBUG)
  endif()
  add_test(NAME ${name} COMMAND test_${name})
endforeach()
